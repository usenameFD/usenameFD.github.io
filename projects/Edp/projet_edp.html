import numpy as np
from scipy.sparse import lil_matrix, csr_matrix
from scipy.sparse.linalg import spsolve
import matplotlib.pyplot as plt

# Parameters
r = 0.5
R = np.sqrt(2)
kappa = 0.1
vs = 0.5
a = 0.2
sigma_x = 0.5
sigma_y = 0.2

M = 80  # Grid size
X_min, X_max = -2.0, 2.0
Y_min, Y_max = -2.0, 2.0
h = (X_max - X_min) / (M + 1)

# Grid points (interior only)
x = X_min + h * np.arange(1, M + 1)
y = Y_min + h * np.arange(1, M + 1)

# Meshgrid
X, Y = np.meshgrid(x, y, indexing='ij')

# Radius at each point
r_node = np.sqrt(X**2 + Y**2)

# Masks for different regions
gamma_r = (r_node <= r)        # Inner boundary
gamma_R = (r_node >= R)        # Outer boundary
omega_int = ~(gamma_r | gamma_R)  # Interior of annulus

# Define functions
def vc_func(x, y):
    return 1.0 - a * np.sin(np.pi * (x**2 + y**2 - r**2) / (R**2 - r**2))

def f_func(x, y):
    # Cas A: u_ex est connu
    k = 0.5 * np.pi/(R**2 - r**2)
    arg = k*(x**2 + y**2 - r**2)
    u_x = 2*k * x * np.cos(arg)
    u_y = 2*k * y * np.cos(arg)
    u_xx = - 4*k * x**2 * np.sin(arg) + 2*k * np.cos(arg)
    u_yy = - 4*k * y**2 * np.sin(arg) + 2*k * np.cos(arg)
    grad_norm_1 = np.abs(u_x) + np.abs(u_y)
    grad_norm_2 = np.sqrt(u_x**2 + u_y**2)
    return -0.5*(sigma_x**2 * u_xx + sigma_y**2 * u_yy) - vc_func(x, y)*u_x + vs*grad_norm_2 - kappa*grad_norm_1

    #return 1.0  # Case B (or modify for Case A)

# For Case A (with exact solution), compute error
def u_exact(x, y):
    r2 = x**2 + y**2
    return np.sin(np.pi/2 * (r2 - r**2) / (R**2 - r**2))


U_exact = u_exact(X, Y)


def newton_smooth(max_iter = 50, tol = 1e-3):
    # Map 2D indices to 1D
    def idx(i, j):
        return i * M + j 
        
    # Compute coefficient matrices on the grid
    V = np.zeros((M, M))
    F = np.zeros((M, M))
    for i in range(M):
        for j in range(M):
            V[i, j] = vc_func(x[i], y[j])
            F[i, j] = f_func(x[i], y[j])

    # First derivative (centered)
    A1 = np.diag(0.5 * np.ones(M-1), 1) - np.diag(0.5 * np.ones(M-1), -1)
    A1[0, :] = 0  # Boundary conditions handled separately
    A1[-1, :] = 0

    # Scale derivatives by grid spacing
    D1 = A1 / (2 * h)  # First derivative operator

    # Initial guess
    U_prec = np.zeros((M, M))
    U_prec[gamma_R] = 1.0  # Outer boundary condition
    U_prec[gamma_r] = 0.0  # Inner boundary condition

    for iteration in range(max_iter):
        print(f"Iteration {iteration+1}")
        
        # Compute derivatives of current solution
        Ux = D1 @ U_prec              # ∂u/∂x
        Uy = U_prec @ D1.T            # ∂u/∂y
        
        # Compute gradient norm (avoid division by zero)
        grad_norm = np.sqrt(Ux**2 + Uy**2)
        grad_norm = np.maximum(grad_norm, 1e-12)  # Regularization
        
        # Alpha coefficients (normalized gradient)
        alpha_x = Ux / grad_norm
        alpha_y = Uy / grad_norm
        
        # Beta coefficients (sign of derivatives)
        beta_x = np.sign(Ux)
        beta_y = np.sign(Uy)
        
        # Build the linear system A @ u_new = b
        # System size: N = M*M
        N = M * M
        A_matrix = lil_matrix((N, N))
        b = np.zeros(N)
        
        # Assemble the system
        for i in range(M):
            for j in range(M):
                I = idx(i, j)

                if gamma_r[i, j] or gamma_R[i, j]:
                    # Boundary nodes: Dirichlet condition
                    A_matrix[I, I] = 1.0
                    if gamma_r[i, j]:
                        b[I] = 0.0
                    else:
                        b[I] = 1.0
                else:
                    # Interior nodes: discretize PDE
                    # u_xx term
                    if i > 0 and i < M-1:
                        A_matrix[I, idx(i+1, j)] += -0.5 * sigma_x**2 * (1/h**2) + (1/(2*h)) * ( -V[i, j] + vs * alpha_x[i, j] - kappa * beta_x[i, j])
                        A_matrix[I, idx(i, j)]   += sigma_x**2 * (1/h**2)
                        A_matrix[I, idx(i-1, j)] += -0.5 * sigma_x**2 * (1/h**2) - (1/(2*h)) * ( -V[i, j] + vs * alpha_x[i, j] - kappa * beta_x[i, j])    
                    
                    # u_yy term
                    if j > 0 and j < M-1:
                        A_matrix[I, idx(i, j+1)] += -0.5 * sigma_y**2 * (1/h**2) +  (1/(2*h)) * (vs * alpha_y[i, j] - kappa * beta_y[i, j])
                        A_matrix[I, idx(i, j)]   += sigma_y**2 * (1/h**2)
                        A_matrix[I, idx(i, j-1)] += -0.5 * sigma_y**2 * (1/h**2) - (1/(2*h)) * ( vs * alpha_y[i, j] - kappa * beta_y[i, j] )
                    
                    # Source term (right-hand side)
                    b[I] = F[i, j]
        
        # Convert to CSR format for efficient solving
        A_csr = A_matrix.tocsr()
        
        # Solve linear system
        u_new_flat = spsolve(A_csr, b)
        U_new = u_new_flat.reshape(M, M)
        
        # Enforce boundary conditions
        U_new[gamma_r] = 0.0
        U_new[gamma_R] = 1.0
        
        # Check convergence
        diff = np.linalg.norm(U_new - U_prec) #/ np.linalg.norm(U_prec + 1e-12)
        print(f"  Absolute change: {diff:.2e}")
        
        if diff < tol:
            print(f"Converged after {iteration+1} iterations")
            U_prec = U_new.copy()
            break
        
        U_prec = U_new.copy()
    return U_prec # Final solution

import torch
import torch.nn as nn

class HJBI_Net(nn.Module):
    """
    Réseau fully-connected pour approcher u(x,y).

    Architecture : [2] → [hidden]*n_layers → [1]
    Activation   : tanh (douce, favorable aux dérivées secondes)
    """
    def __init__(self, hidden=64, n_layers=4):
        super().__init__()
        layers = [nn.Linear(2, hidden), nn.Tanh()]
        for _ in range(n_layers - 1):
            layers += [nn.Linear(hidden, hidden), nn.Tanh()]
        layers += [nn.Linear(hidden, 1)]
        self.net = nn.Sequential(*layers)

    def forward(self, xy):
        return self.net(xy).squeeze(-1)


def pde_residual(net, xy, case='B'):
    """
    Calcule le résidu de l'EDP en un batch de points xy ∈ Ω.
    Utilise la différentiation automatique de PyTorch.
    """
    xy.requires_grad_(True)
    u = net(xy)

    # Gradient de u par rapport à (x, y)
    grads = torch.autograd.grad(u.sum(), xy, create_graph=True)[0]
    u_x, u_y = grads[:, 0], grads[:, 1]

    # Dérivées secondes
    u_xx = torch.autograd.grad(u_x.sum(), xy, create_graph=True)[0][:, 0]
    u_yy = torch.autograd.grad(u_y.sum(), xy, create_graph=True)[0][:, 1]

    x_c = xy[:, 0]
    y_c = xy[:, 1]

    # Vent vc(x,y)
    xi_c = (x_c**2 + y_c**2 - r**2) / (R**2 - r**2)
    vc_c = 1.0 - a * torch.sin(np.pi * xi_c)

    # Termes non-linéaires
    eps    = 1e-8
    grad2  = torch.sqrt(u_x**2 + u_y**2 + eps)
    grad1  = torch.abs(u_x) + torch.abs(u_y)

    # Résidu EDP
    res = (-0.5 * (sigma_x**2 * u_xx + sigma_y**2 * u_yy)
            - vc_c * u_x
            + vs * grad2
            - kappa * grad1)

    if case == 'A':
        # f = f_manufactured calculé en numpy puis converti
        xy_np = xy.detach().cpu().numpy()
        f_val = torch.tensor(
            f_func(xy_np[:, 0], xy_np[:, 1]),
            dtype=torch.float32, device=xy.device
        )
    else:
        f_val = torch.ones_like(res)   # f = 1

    return res - f_val


def sample_domain(n_pde=2000, n_bc=500, device='cpu'):
    """Génère des points d'entraînement dans Ω et sur ∂Ω."""
    # Points intérieurs (rejection sampling)
    pts = []
    while len(pts) < n_pde:
        xy = (np.random.rand(n_pde * 4, 2) * 2 * R) - R
        rr = np.sum(xy**2, axis=1)
        mask = (rr >= r**2) & (rr <= R**2)
        pts.append(xy[mask])
    pts_pde = torch.tensor(
        np.vstack(pts)[:n_pde], dtype=torch.float32, device=device
    )

    # Points sur ∂Br (bord intérieur)
    theta_in = np.linspace(0, 2 * np.pi, n_bc, endpoint=False)
    bc_in = r * np.stack([np.cos(theta_in), np.sin(theta_in)], axis=1)
    pts_bc_in = torch.tensor(bc_in, dtype=torch.float32, device=device)

    # Points sur ∂BR (bord extérieur)
    theta_out = np.linspace(0, 2 * np.pi, n_bc, endpoint=False)
    bc_out = R * np.stack([np.cos(theta_out), np.sin(theta_out)], axis=1)
    pts_bc_out = torch.tensor(bc_out, dtype=torch.float32, device=device)

    return pts_pde, pts_bc_in, pts_bc_out


def train_pinns(case='B', n_epochs=5000, lr=1e-3, hidden=64, n_layers=4,
                n_pde=3000, n_bc=500, verbose=True):
    """
    Entraîne le réseau PINN pour le problème Zermelo.

    Retourne le réseau entraîné et l'historique des pertes.
    """
    device = 'cpu'
    net    = HJBI_Net(hidden=hidden, n_layers=n_layers).to(device)
    optim  = torch.optim.Adam(net.parameters(), lr=lr)
    sched  = torch.optim.lr_scheduler.StepLR(optim, step_size=2000, gamma=0.5)

    loss_hist = []
    pts_pde, pts_in, pts_out = sample_domain(n_pde, n_bc, device)

    for epoch in range(n_epochs):
        optim.zero_grad()

        # Résidu PDE
        res = pde_residual(net, pts_pde.clone(), case=case)
        loss_pde = (res**2).mean()

        # CL intérieur : u = 0 sur ∂Br
        u_in  = net(pts_in)
        loss_in  = (u_in**2).mean()

        # CL extérieur : u = 1 sur ∂BR
        u_out = net(pts_out)
        loss_out = ((u_out - 1.0)**2).mean()

        loss = loss_pde + (loss_in + loss_out)
        loss.backward()
        optim.step()
        sched.step()

        loss_hist.append(loss.item())

        if verbose and epoch % 500 == 0:
            print(f"  Epoch {epoch:5d}: loss = {loss.item():.4e}  "
                    f"(pde={loss_pde.item():.2e}, bc={loss_in.item()+loss_out.item():.2e})")

    return net, loss_hist


def eval_pinns_on_grid(net, M=80):
    """Évalue le réseau sur une grille M×M pour visualisation."""
    hx = 4.0 / (M + 1)
    x_arr = np.linspace(-2 + hx, 2 - hx, M)
    y_arr = np.linspace(-2 + hx, 2 - hx, M)
    X, Y = np.meshgrid(x_arr, y_arr, indexing='ij')

    xy_flat = np.stack([X.ravel(), Y.ravel()], axis=1).astype(np.float32)
    with torch.no_grad():
        u_flat = net(torch.tensor(xy_flat)).numpy()

    U = u_flat.reshape(M, M)
    # Appliquer les CL exactes sur les noeuds de bord
    r2 = X**2 + Y**2
    U[r2 <= r**2] = 0.0
    U[r2 >= R**2] = 1.0

    return X, Y, U

def pde_residual(net, xy, case='B', h_fd=1e-3):
    """
    Calcule le résidu de l'EDP via différences finies sur le réseau.
    xy : tensor (n_pts, 2), points collocation dans Ω
    """
    # Déplacements unitaires
    ex = torch.zeros_like(xy)
    ex[:, 0] = h_fd
    ey = torch.zeros_like(xy)
    ey[:, 1] = h_fd

    # Évaluations du réseau aux points décalés (no_grad : pas de backprop à travers les FD)
    with torch.no_grad():
        u_c  = net(xy)           # u(x, y)
        u_xp = net(xy + ex)     # u(x+h, y)
        u_xm = net(xy - ex)     # u(x-h, y)
        u_yp = net(xy + ey)     # u(x, y+h)
        u_ym = net(xy - ey)     # u(x, y-h)

    # Dérivées premières (centrées, ordre 2)
    u_x = (u_xp - u_xm) / (2 * h_fd)
    u_y = (u_yp - u_ym) / (2 * h_fd)

    # Dérivées secondes (ordre 2)
    u_xx = (u_xp - 2 * u_c + u_xm) / h_fd**2
    u_yy = (u_yp - 2 * u_c + u_ym) / h_fd**2

    x_c = xy[:, 0]
    y_c = xy[:, 1]

    # Vent vc(x, y)
    xi_c = (x_c**2 + y_c**2 - r**2) / (R**2 - r**2)
    vc_c = 1.0 - a * torch.sin(torch.tensor(np.pi) * xi_c)

    # Termes non-linéaires
    eps   = 1e-8
    grad2 = torch.sqrt(u_x**2 + u_y**2 + eps)
    grad1 = torch.abs(u_x) + torch.abs(u_y)

    # Résidu EDP
    res = (-0.5 * (sigma_x**2 * u_xx + sigma_y**2 * u_yy)
           - vc_c * u_x
           + vs * grad2
           - kappa * grad1)

    if case == 'A':
        xy_np = xy.detach().cpu().numpy()
        f_val = torch.tensor(
            f_func(xy_np[:, 0], xy_np[:, 1]),
            dtype=torch.float32, device=xy.device
        )
    else:
        f_val = torch.ones_like(res)

    return res - f_val
