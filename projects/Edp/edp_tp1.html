import pandas as pd

# ============================================================================
# 1. PRÉPARATION DES DONNÉES
# ============================================================================


df['date'] = pd.to_datetime(df['date'])

# ============================================================================
# 2. CALCUL DU % CHANGE ET DES TRADES
# ============================================================================

# Calculer le pourcentage de changement
df['% Change'] = df['close'].pct_change().fillna(0)

# Calculer Absolute Integer Change selon les règles
df['Abs_Integer_Change'] = (abs(df['% Change']) * 100).round(0).astype(int)

# Déterminer Is Trade: True si Abs_Integer_Change est pair et > 0
df['Is Trade'] = (df['Abs_Integer_Change'] > 0) & ((df['Abs_Integer_Change'] % 2 == 0) | ((df['Abs_Integer_Change'] - 1) % 3 == 0))


#df['Is Trade'] = (df['Abs_Integer_Change'] > 0) & (df['Abs_Integer_Change'] % 2 == 0)

# ============================================================================
# 3. RÉPONSE À LA QUESTION 3: Nombre total de trades
# ============================================================================

total_trades = df['Is Trade'].sum()
print(f"3. Nombre total de trades: {total_trades}")

# ============================================================================
# 4. RÉPONSE À LA QUESTION 4: Plus grand nombre de jours consécutifs sans trade
# ============================================================================

# Trouver les séquences sans trade
df['Trade Group'] = (df['Is Trade'] != df['Is Trade'].shift()).cumsum()

# Compter les jours consécutifs sans trade
no_trade_sequences = df[~df['Is Trade']].groupby('Trade Group').size()
max_consecutive_no_trade = no_trade_sequences.max() if not no_trade_sequences.empty else 0

print(f"4. Plus grand nombre de jours consécutifs sans trade: {max_consecutive_no_trade}")

# ============================================================================
# 5. CALCUL DES PROFITS ET RÉPONSE À LA QUESTION 5
# ============================================================================

# Calculer le profit pour chaque jour de trade
profit = [0.0]
for i in range(len(df) - 1):
    if df.iloc[i]['Is Trade']:
        buy_price = df.iloc[i]['close']
        sell_price = df.iloc[i + 1]['close']
        profit.append((sell_price - buy_price) / buy_price)
    else:
        profit.append(0.0)

# Calculer le profit moyen par trade
df['Profit'] = profit
trade_profits = df[df['Is Trade']]['Profit']
avg_profit = trade_profits.mean() * 100 if not trade_profits.empty else 0
avg_profit_rounded = round(avg_profit, 2)

print(f"5. Profit moyen par trade: {avg_profit_rounded:.2f}%")

# ============================================================================
# 6. CALCUL DES PROFITS TOTAUX ET RÉPONSE À LA QUESTION 6
# ============================================================================

# Calculer les profits totaux cumulés
df['Total Profit'] = df['Profit'].cumsum() * 100  # En pourcentage

# Profit total final
total_profit_final = df['Total Profit'].iloc[-1]
total_profit_rounded = round(total_profit_final, 2)

print(f"6. Profit total de la stratégie: {total_profit_rounded:.2f}%")

# ============================================================================
# 7. RÉPONSE À LA QUESTION 7: Dates des profits totaux max et min
# ============================================================================

# Trouver les dates des profits totaux maximum et minimum
max_profit_idx = df['Total Profit'].idxmax()
min_profit_idx = df['Total Profit'].idxmin()

max_profit_date = df.loc[max_profit_idx, 'date'].strftime('%Y-%m-%d')
min_profit_date = df.loc[min_profit_idx, 'date'].strftime('%Y-%m-%d')

print(f"\n7. Dates des profits totaux max et min:")
print(f"   Maximum profit date: {max_profit_date}")
print(f"   Minimum profit date: {min_profit_date}")

# Vérifier les dates fournies
check_dates_q7 = [
    '2020-06-01', '2021-01-10', '2020-12-01',
    '2020-06-12', '2020-11-30', '2021-05-10'
]

print("\n   Checking which dates match:")
for date_str in check_dates_q7:
    if date_str == max_profit_date or date_str == min_profit_date:
        print(f"   ✓ {date_str}")
    else:
        print(f"   ✗ {date_str}")

# ============================================================================
# 8. CALCUL DU MAXIMUM DRAWDOWN ET RÉPONSE À LA QUESTION 8
# ============================================================================

# Calculer le maximum drawdown
df['Running Max'] = df['Total Profit'].cummax()
df['Drawdown'] = df['Total Profit'] - df['Running Max']

# Le maximum drawdown est la valeur la plus négative (ou la plus petite)
max_drawdown = abs(df['Drawdown'].min())
max_drawdown_rounded = round(max_drawdown, 2)

print(f"\n8. Maximum Drawdown: {max_drawdown_rounded:.2f}%")
